  %% ================================================
  %% SANCTIONS — Modul 1: izricanje sankcija
  %% ================================================
  namespace sanctions {
    class Driver {
      +String id [JMBG]
      +String firstName
      +String lastName
      +LocalDate dateOfBirth
      +DrivingLicense license
      +isProbationary() boolean
      +getFullName() String
    }
    class DrivingLicense {
      +String licenseNumber
      +LocalDate issuedAt
      +LicenseType type
      +Set~LicenseCategory~ categories
      +covers(VehicleCategory) boolean
    }
    class LicenseType {
      <<enumeration>>
      PERMANENT
      PROBATIONARY
    }
    class LicenseCategory {
      <<enumeration>>
      AM A1 A2 A
      B1 B BE
      C1 C CE
      D1 D DE
      +getVehicleCategory() VehicleCategory
    }
    class LicenseRevocation {
      +Long id
      +Driver driver
      +LocalDate revokedAt
      +int activePointsAtRevocation
      +boolean probationary
      +String reason
    }
    class PointPenalty {
      <<event @Role @Timestamp @Expires 730d>>
      +Long id
      +Driver driver
      +int points
      +Date issuedAt
      +String externalId
    }
    class ActivePointsSnapshot {
      +Driver driver
      +int activePoints
      +LocalDate computedAt
    }
    class Violation {
      +Driver driver
      +Vehicle vehicle
      +ViolationType type
      +Location location
      +int speedOverLimitKmH
      +double bloodAlcoholLevel
    }
    class ViolationContext {
      +Violation violation
      +boolean causedAccident
      +boolean injuredPersons
      +boolean pedestrianIsChild
    }
    class ViolationType {
      <<enumeration>>
      SPEEDING ALCOHOL
      RED_LIGHT STOP_SIGN_IGNORED
      OVERTAKING_SOLID_LINE ...
      NO_LICENSE DRIVING_DURING_BAN
      FLEEING_SCENE ...
      ~40 vrednosti
    }
    class Location {
      <<enumeration>>
      URBAN
      OPEN_ROAD
      SCHOOL_ZONE
      ANY
    }
    class Sanction {
      +Violation violation
      +String lawArticle
      +Severity severity
      +int minFine / maxFine
      +int baseFine / finalFine
      +int basePoints / finalPoints
      +int drivingBanDays
      +int prisonDaysMin / prisonDaysMax
      +int communityServiceHoursMin / Max
      +boolean alternativePenalty
      +List~String~ explanation
      +builder() Builder
    }
    class SanctionSummary {
      +Driver driver
      +List~Sanction~ sanctions
      +int totalFineMin / totalFineMax
      +int totalPoints
      +int drivingBanMinDays / drivingBanMaxDays
      +int prisonMinDays / prisonMaxDays
      +List~String~ overallExplanation
    }
    class Severity {
      <<enumeration>>
      VIOLENT (čl. 329)
      SEVERE (čl. 330)
      MAJOR (čl. 331)
      MODERATE (čl. 332)
      PETTY (čl. 332a)
      MINOR (čl. 333)
      MINIMAL (čl. 334)
    }
  }

  namespace fault {
    class Accident {
      +String id
      +LocalDateTime occurredAt
      +AccidentScene scene
      +List~Participant~ participants
      +boolean injuredPersons
      +boolean fatalOutcome
      +int materialDamageRsd
    }
    class AccidentScene {
      +SignalizationType signalization
      +Visibility visibility
      +WeatherCondition weather
      +RoadCondition roadCondition
      +int speedLimitKmH
      +boolean intersection
    }
    class Participant {
      +String id
      +Driver driver
      +Vehicle vehicle
      +Maneuver maneuver
      +int speedKmH
      +double distanceToVehicleAhead
      +boolean underAlcoholInfluence
      +boolean ignoredRedLight
    }
    class ParticipantAction {
      +Participant participant
      +ParticipantActionType type
      +LocalDateTime timestamp
    }

      +Participant participant
      +int percentage
      +BlameType type
      +List~String~ reasoning
